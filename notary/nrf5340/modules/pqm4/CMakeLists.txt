# Zephyr module wrapper for pqm4 ML-DSA-44.
# Tested against mupq/pqm4 e1146b1bc180c66689682f6ce63bd8f126a51ace (mupq submodule 3bce0c95e9ae11d97daa7285f84bf2e1a90ad451).

if(CONFIG_PQM4_MLDSA44)
  if(NOT PQM4_DIR)
    if(DEFINED ENV{PQM4_DIR})
      set(PQM4_DIR $ENV{PQM4_DIR})
    elseif(DEFINED WEST_TOPDIR)
      set(PQM4_DIR ${WEST_TOPDIR}/modules/crypto/pqm4)
    else()
      get_filename_component(PQM4_DIR ${ZEPHYR_BASE}/../modules/crypto/pqm4 ABSOLUTE)
    endif()
  endif()
  set(PQM4_DIR ${PQM4_DIR} CACHE PATH "pqm4 checkout (https://github.com/mupq/pqm4)")

  if(CONFIG_PQM4_MLDSA44_M4FSTACK)
    set(impl ${PQM4_DIR}/crypto_sign/ml-dsa-44/m4fstack)
  else()
    set(impl ${PQM4_DIR}/crypto_sign/ml-dsa-44/m4f)
  endif()

  if(NOT EXISTS ${impl}/sign.c)
    message(FATAL_ERROR "pqm4 ML-DSA-44 sources not found at ${impl}. Clone https://github.com/mupq/pqm4 "
                        "and set PQM4_DIR (env or -DPQM4_DIR=...).")
  endif()
  if(NOT EXISTS ${PQM4_DIR}/mupq/common/fips202.c)
    message(FATAL_ERROR "${PQM4_DIR}/mupq is empty: run 'git -C ${PQM4_DIR} submodule update --init mupq'.")
  endif()

  file(GLOB impl_sources ${impl}/*.c ${impl}/*.S ${impl}/*.s)

  zephyr_library_named(pqm4_mldsa44)
  zephyr_library_sources(
    ${impl_sources}
    ${PQM4_DIR}/mupq/common/fips202.c
    ${PQM4_DIR}/common/keccakf1600.S
  )
  if(CONFIG_PQM4_RANDOMBYTES)
    zephyr_library_sources(zephyr/randombytes_zephyr.c)
  endif()
  # shim/ first: it provides randombytes.h (the mupq copy is a symlink into the
  # pqclean submodule, which we do not need).
  zephyr_library_include_directories(shim ${impl} ${PQM4_DIR}/mupq/common ${PQM4_DIR}/common)
  zephyr_library_compile_definitions(PQM4)
  if(CONFIG_PQM4_OPTIMIZE_SPEED)
    zephyr_library_compile_options($<$<COMPILE_LANGUAGE:C>:-O3>)
  endif()

  # Public headers: <pqm4/mldsa44_api.h> includes the implementation's api.h by absolute
  # path, so params.h/config.h etc. resolve next to it and do not leak into the global
  # include path.
  set(gen_inc ${CMAKE_CURRENT_BINARY_DIR}/include)
  file(WRITE ${gen_inc}/pqm4/mldsa44_api.h.tmp
       "/* Generated by modules/pqm4/CMakeLists.txt */\n#include \"${impl}/api.h\"\n")
  configure_file(${gen_inc}/pqm4/mldsa44_api.h.tmp ${gen_inc}/pqm4/mldsa44_api.h COPYONLY)
  zephyr_include_directories(${gen_inc} include)
endif()
